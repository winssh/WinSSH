# Ctrl+C at the password prompt stops WinSSH cleanly (no hang, exit 255).
# Use a user name the server does not know, so key auth fails and the password prompt appears:
#   .\tests\e2e\run.ps1 -Target winssh_no_such_user@10.0.0.111 -Script tests\e2e\password-cancel.txt
expect password:
send \x03
expect interrupted
exit 255
